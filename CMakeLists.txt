cmake_minimum_required(VERSION 3.21)
project(openbrush_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(LibArchive REQUIRED)
find_package(Threads REQUIRED)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
add_library(brush-core src/document.cpp src/tilt.cpp src/mesh.cpp src/brush.cpp src/editor.cpp src/navigation.cpp src/controller-profiles.cpp src/save-worker.cpp src/gltf.cpp src/reference.cpp)
target_include_directories(brush-core PUBLIC src)
target_link_libraries(brush-core PUBLIC LibArchive::LibArchive Threads::Threads PNG::PNG JPEG::JPEG)
target_compile_options(brush-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(openbrush-tool src/tool.cpp)
target_link_libraries(openbrush-tool PRIVATE brush-core)
include(CTest)
if(BUILD_TESTING)
  add_executable(brush-core-test tests/core.cpp)
  target_compile_options(brush-core-test PRIVATE -UNDEBUG)
  target_link_libraries(brush-core-test PRIVATE brush-core)
  add_test(NAME brush-core COMMAND brush-core-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/upstream-one-stroke.tilt)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(XR REQUIRED IMPORTED_TARGET openxr)
pkg_check_modules(GRAPHICS REQUIRED IMPORTED_TARGET egl gl)
add_library(brush-render src/render.cpp src/graphics.cpp src/reference-render.cpp)
target_link_libraries(brush-render PUBLIC brush-core PkgConfig::GRAPHICS)
add_executable(openbrush-native src/main.cpp src/xr.cpp)
target_link_libraries(openbrush-native PRIVATE brush-render PkgConfig::XR)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME native-render COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/render.py $<TARGET_FILE:openbrush-native> ${CMAKE_CURRENT_BINARY_DIR}/native-demo.ppm)
  add_test(NAME tilt-interoperability COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/interop.py $<TARGET_FILE:openbrush-tool> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/upstream-one-stroke.tilt)
endif()
option(OPENBRUSH_ENABLE_X11 "Allow the explicitly selected X11 FLTK backend" OFF)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
pkg_check_modules(FONTS REQUIRED IMPORTED_TARGET fontconfig zlib)
target_sources(openbrush-native PRIVATE fltk/main.cpp fltk/studio.cpp fltk/canvas.cpp)
target_include_directories(openbrush-native PRIVATE fltk)
target_compile_definitions(openbrush-native PRIVATE OPENBRUSH_ENABLE_X11=$<BOOL:${OPENBRUSH_ENABLE_X11}>)
target_link_libraries(openbrush-native PRIVATE fltk::gl-shared fltk::fltk-shared fltk::images-shared PkgConfig::FONTS)
include(fltk/desktop-menu.cmake)
de_xr_desktop_menu(openbrush-native)
if(BUILD_TESTING)
  add_executable(brush-editor-test tests/editor.cpp)
  target_link_libraries(brush-editor-test PRIVATE brush-core)
  target_compile_options(brush-editor-test PRIVATE -UNDEBUG)
  add_test(NAME brush-editor COMMAND brush-editor-test)
  if(OPENBRUSH_ENABLE_X11)
    add_executable(brush-studio-test tests/studio.cpp fltk/studio.cpp fltk/canvas.cpp src/xr.cpp)
    target_include_directories(brush-studio-test PRIVATE fltk)
    target_compile_definitions(brush-studio-test PRIVATE OPENBRUSH_ENABLE_X11=1)
    target_link_libraries(brush-studio-test PRIVATE brush-render PkgConfig::XR fltk::gl-shared fltk::fltk-shared fltk::images-shared PkgConfig::FONTS)
    de_xr_desktop_menu(brush-studio-test)
    add_test(NAME brush-studio COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/studio.py $<TARGET_FILE:brush-studio-test> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/upstream-one-stroke.tilt ${CMAKE_CURRENT_BINARY_DIR}/studio-ui.ppm)
  endif()
endif()
if(BUILD_TESTING AND NOT OPENBRUSH_ENABLE_X11)
  add_test(NAME wayland-default COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/backend.py $<TARGET_FILE:openbrush-native>)
endif()
pkg_check_modules(PALETTE_FONTS REQUIRED IMPORTED_TARGET freetype2 fontconfig)
add_library(brush-palette src/palette.cpp src/interaction.cpp)
target_link_libraries(brush-palette PUBLIC brush-core PkgConfig::PALETTE_FONTS)
target_sources(brush-render PRIVATE src/panel-render.cpp)
target_link_libraries(brush-render PUBLIC brush-palette)
if(BUILD_TESTING)
  add_executable(brush-interaction-test tests/interaction.cpp)
  target_compile_options(brush-interaction-test PRIVATE -UNDEBUG)
  target_link_libraries(brush-interaction-test PRIVATE brush-palette)
  add_test(NAME xr-interaction COMMAND brush-interaction-test)
endif()
if(BUILD_TESTING)
  add_executable(brush-palette-render-test tests/palette-render.cpp)
  target_compile_options(brush-palette-render-test PRIVATE -UNDEBUG)
  target_link_libraries(brush-palette-render-test PRIVATE brush-render)
  add_test(NAME palette-gpu COMMAND brush-palette-render-test ${CMAKE_CURRENT_BINARY_DIR}/palette-bitmap.ppm)
endif()
if(BUILD_TESTING)
  add_executable(brush-palette-appearance-test tests/palette-appearance.cpp)
  target_compile_options(brush-palette-appearance-test PRIVATE -UNDEBUG)
  target_link_libraries(brush-palette-appearance-test PRIVATE brush-palette)
  add_test(NAME palette-appearance COMMAND brush-palette-appearance-test)
endif()
if(BUILD_TESTING)
  add_executable(brush-touch-test tests/touch.cpp)
  target_compile_options(brush-touch-test PRIVATE -UNDEBUG)
  target_link_libraries(brush-touch-test PRIVATE brush-palette)
  add_test(NAME touch-controls COMMAND brush-touch-test)
endif()
if(BUILD_TESTING)
  add_executable(brush-save-worker-test tests/save-worker.cpp)
  target_compile_options(brush-save-worker-test PRIVATE -UNDEBUG)
  target_link_libraries(brush-save-worker-test PRIVATE brush-core)
  add_test(NAME background-save COMMAND brush-save-worker-test)
endif()

include(GNUInstallDirs)
set_target_properties(openbrush-native PROPERTIES
  INSTALL_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>")
install(TARGETS openbrush-native openbrush-tool RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/org.domed.OpenBrushNative.desktop
  DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/org.domed.OpenBrushNative.svg
  DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES packaging/org.domed.OpenBrushNative.xml
  DESTINATION ${CMAKE_INSTALL_DATADIR}/mime/packages)
install(FILES README.md LICENSE NOTICE third_party/upstream.json
  third_party/ui-adapters.json third_party/LICENSE.native-graphics
  DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/openbrush-native)
if(BUILD_TESTING)
  add_executable(brush-gltf-fixture tests/gltf.cpp)
  target_compile_options(brush-gltf-fixture PRIVATE -UNDEBUG)
  target_link_libraries(brush-gltf-fixture PRIVATE brush-core)
  add_test(NAME gltf-export COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/gltf.py
    $<TARGET_FILE:brush-gltf-fixture> $<TARGET_FILE:openbrush-tool> $<TARGET_FILE:openbrush-native>
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/upstream-one-stroke.tilt)
endif()
if(BUILD_TESTING)
  add_executable(brush-reference-test tests/reference.cpp)
  target_compile_options(brush-reference-test PRIVATE -UNDEBUG)
  target_link_libraries(brush-reference-test PRIVATE brush-render)
  add_test(NAME reference-images COMMAND brush-reference-test)
endif()
